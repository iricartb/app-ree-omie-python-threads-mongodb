// This function is the endpoint's request handler.
exports = function({ query, headers, body}, response) {
  try {
    // Allow values
    const collectionAllowValues    = [ "codsvbaj", "codsvsub", "compodem_breakdown", "compodem", "losses", "periods", "scdsvdem", "sphdem_dd", "sphven_dd", "sprpcap" ];
    const liquidationAllowValues   = [ "BEST", "A1", "A2", "A3", "A4", "A5", "A6", "A7", "A8", "A9", "C1", "C2", "C3", "C4", "C5", "C6", "C7", "C8", "C9" ];
    const typeDemandAllowValues    = [ "CUR", "NOCUR", "TOTAL" ];
    const tariffNameAllowValues    = [ "2.0TD", "3.0TD", "3.0TDVE", "6.1TD", "6.1TDVE", "6.2TD", "6.3TD", "6.4TD" ];
    const subsystemNameAllowValues = [ "BALEARES", "CANARIAS", "CEUTA", "MELILLA", "PENINSULA" ];
    const hourAllowValues          = [ "hourly", "quarterhourly" ];
    const pricesAllowValues        = [ "kWh", "MWh" ];
    
    let jsonBody = JSON.parse(body.text());
    
    // Collection
    let collection = jsonBody.collection;
    
    // Filters
    if (jsonBody.filters == undefined) jsonBody.filters = {};
    
    let filterStartDate     = jsonBody.filters.start_date;
    let filterEndDate       = jsonBody.filters.end_date;
    let filterLiquidation   = jsonBody.filters.liquidation;
    let filterHour          = jsonBody.filters.hour;
    let filterQuarter       = jsonBody.filters.quarter;
    let filterSegment       = jsonBody.filters.segment;
    let filterTypeDemand    = jsonBody.filters.type_demand;
    let filterTariffId      = jsonBody.filters.tariff_id;
    let filterSubsystemId   = jsonBody.filters.subsystem_id;
    let filterTariffName    = jsonBody.filters.tariff_name;
    let filterSubsystemName = jsonBody.filters.subsystem_name;
    
    // Formats
    if (jsonBody.formats == undefined) jsonBody.formats = {};
    
    let formatHour   = jsonBody.formats.hour;
    let formatPrices = jsonBody.formats.prices;
    
    // Sorters
    let sorters = jsonBody.sorters;
    
    // Limit
    let limit = jsonBody.limit;
    
    // Validations
    if (collection === undefined) { throw new Error("Parámetro collection requerido"); }
    if (!collectionAllowValues.includes(collection)) { throw new Error("El parámetro collection debe contener alguno de estos valores: " + collectionAllowValues.join(", ")); }
    if ((filterStartDate !== undefined) && (isNaN(Date.parse(filterStartDate)))) { throw new Error("El filtro start_date debe ser una fecha válida (YYYY-mm-dd)"); }
    if ((filterEndDate !== undefined) && (isNaN(Date.parse(filterEndDate)))) { throw new Error("El filtro end_date debe ser una fecha válida (YYYY-mm-dd)"); }
    if ((filterTariffName !== undefined) && (!tariffNameAllowValues.includes(filterTariffName))) { throw new Error("El filtro tariff_name debe contener alguno de estos valores: "  + tariffNameAllowValues.join(", ")); }
    if ((filterSubsystemName !== undefined) && (!subsystemNameAllowValues.includes(filterSubsystemName))) { throw new Error("El filtro subsystem_name debe contener alguno de estos valores: "  + subsystemNameAllowValues.join(", ")); }
    if ((filterTypeDemand !== undefined) && (!typeDemandAllowValues.includes(filterTypeDemand))) { throw new Error("El filtro type_demand debe contener alguno de estos valores: "  + typeDemandAllowValues.join(", ")); }
    if ((formatHour !== undefined) && (!hourAllowValues.includes(formatHour.toLowerCase()))) { throw new Error("El formato hour debe contener alguno de estos valores: "  + hourAllowValues.join(", ")); }
    if ((formatPrices !== undefined) && (!pricesAllowValues.includes(formatPrices))) { throw new Error("El formato prices debe contener alguno de estos valores: "  + pricesAllowValues.join(", ")); }
    
    // Database
    const mongodb = context.services.get("MarketsCluster0");
    const db = mongodb.db("ree");
    
    collection = "files_" + collection;

    if (formatHour.toLowerCase() == "quarterhourly") { collection += "_qh"; }
    
    // Filters
    let filters = {}
    
    if (filterStartDate) {
      filters.date = { $gte: filterStartDate };
    }
    
    if (filterEndDate) {
      if (!filters.date) {
        filters.date = {};
      }
      
      filters.date.$lte = filterEndDate;
    }

    if (filterLiquidation)   { filters.liquidation = filterLiquidation; }
    if (filterHour)          { filters.hour = filterHour; }
    if (filterQuarter)       { filters.quarter = filterQuarter; }
    if (filterSegment)       { filters.segment = filterSegment; }
    if (filterTypeDemand)    { filters.type_demand = filterTypeDemand; }
    if (filterTariffId)      { filters.tariff_id = filterTariffId; }
    if (filterSubsystemId)   { filters.subsystem_id = filterSubsystemId; }
    if (filterTariffName)    { filters.tariff_name = filterTariffName; }
    if (filterSubsystemName) { filters.subsystem_name = filterSubsystemName; }

    let documents = db.collection(collection).find(filters);
    
    // Sorters
    if (sorters !== undefined) {
      for (let key in sorters) {
          if (sorters[key] === "ASC") { sorters[key] = 1; }
          else if (sorters[key] === "DESC") { sorters[key] = -1; }
      }
      
      documents = documents.sort(sorters);
    }
    
    if (limit !== undefined) { documents = documents.limit(limit); }
    
    return documents;
  } catch (error) {
    return {
      code: 400,
      error: error.message 
    };
  }
};
